<!-- annotated:92dbf23e3c7735 -->

```cpp
  int popQueue(int st[3], int &dir)
  {
    if (head == nullptr) {
      return 0;
    }

    st[0] = head->x;
    st[1] = head->y;
    st[2] = head->z;
    dir = (int)(head->dir);

    gridQueueEle *temp = head;
    head = head->next;
    delete temp;

    if (head == nullptr) {
      tail = nullptr;
    }
    numEles--;

    return 1;
  }
```
